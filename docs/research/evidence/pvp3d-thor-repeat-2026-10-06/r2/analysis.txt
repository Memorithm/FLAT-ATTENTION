import json,pathlib,collections,statistics,hashlib,sys
root=pathlib.Path(sys.argv[1]); cases=[]; skips=None; checksums={}; runs=[]; ownership=[]
for block in range(1,4):
 b=root/f"block-{block}"
 assert "reservation_released_exit=0" in (b/"control.log").read_text()
 assert "block_complete=5_invocations" in (b/"control.log").read_text()
 assert "PVP_ANF_BANK_COMPLETE,cases=24,candidate_checks=96" in (b/"anf-bank.log").read_text()
 states=(b/"restored-services.txt").read_text()
 assert states.count("ActiveState=active")==3 and states.count("SubState=running")==3
 assert states.count("RefuseManualStart=no")==3 and states.count("Restart=on-failure")==3
 obs=[json.loads(l) for l in (b/"occupancy.jsonl").read_text().splitlines()]
 tags=(b/"occupancy-times.tsv").read_text().splitlines()
 decisions=(b/"occupancy-decisions.txt").read_text().splitlines()
 assert len(obs)==len(tags)==len(decisions)
 for o,t,d in zip(obs,tags,decisions):
  phase=t.split("\t")[0]
  if phase=="measuring":
   assert not o["scan_limit_reached"] and not o["gap_details_truncated"]
   if o["status"]=="partial":
    assert d.startswith("observation=partial_owned_descriptor_churn")
    assert sum(g["count"] for g in o["gaps"])==o["unreadable_entries"]
    assert all(g["stage"]=="fd_metadata" and g["process_identity_verified"] and g["cgroup_v2"]==f"/system.slice/remoteops-pvp-r2-{root.name.split('.')[-1]}-b{block}.service" for g in o["gaps"])
   else: assert d=="observation=observed" and o["unreadable_entries"]==0
  ownership.append({"block":block,"phase":phase,"status":o["status"],"decision":d})
 for invocation in range(1,6):
  log=b/f"run-{invocation}"/"pvp3d.log"; text=log.read_text()
  for marker in ["source_revision=50c0230cc645bd44f1520e2c7194878a4b13009c","adapter=NVIDIA Tegra NVIDIA Thor","backend=Vulkan","hardware_measurement_eligible=true","qualification_status=complete","performance_claim=none","candidate_order=ABC_ACB_BAC_BCA_CAB_CBA"]:
   assert marker in text,(log,marker)
  rows={}; samples=collections.defaultdict(dict); rejected={}
  for line in text.splitlines():
   p=line.split(",")
   if p[0]=="measured":
    key=(int(p[1]),int(p[2]),p[3]); assert key not in rows
    assert len(p)==15 and p[6:8]==["5","30"] and p[14]=="none"
    rows[key]=p
   elif p[0]=="sample":
    key=(int(p[1]),int(p[2]),p[3]); idx=int(p[4])
    assert idx not in samples[key]; samples[key][idx]=int(p[5])
   elif p[0]=="skipped_limit":
    key=(int(p[1]),int(p[2]),p[3]); assert key not in rejected; rejected[key]=p[13]
  assert len(rows)+len(rejected)==63
  assert len(rows)==45 and len(rejected)==18
  assert set(samples)==set(rows)
  for key,p in rows.items():
   ss=samples[key]; assert set(ss)==set(range(30))
   assert all(v>0 for v in ss.values())
   ordered=sorted(ss.values()); assert [int(p[9]),int(p[10])]==[ordered[14],ordered[28]]
   cohort=key[:2]
   checksum=tuple(p[11:13])
   if cohort in checksums: assert checksum==checksums[cohort]
   checksums[cohort]=checksum
  if skips is None: skips=rejected
  else: assert skips==rejected
  geoms=sorted({k[:2] for k in rows})
  assert len(geoms)==15
  for kg in geoms:
   assert {k[2] for k in rows if k[:2]==kg}=={"vec4","fused2","tile8"}
   p50={c:int(rows[(*kg,c)][9]) for c in ["vec4","fused2","tile8"]}
   p95={c:int(rows[(*kg,c)][10]) for c in ["vec4","fused2","tile8"]}
   cases.append({"block":block,"invocation":invocation,"k":kg[0],"g":kg[1],"p50_ns":p50,"p95_ns":p95})
  for kg in {k[:2] for k in rejected}:
   assert {k[2] for k in rejected if k[:2]==kg}=={"vec4","fused2","tile8"}
  runs.append({"block":block,"invocation":invocation,"measured_rows":len(rows),"skipped_rows":len(rejected),"raw_samples":sum(len(v) for v in samples.values())})
panels=[]
for kg in sorted({(r["k"],r["g"]) for r in cases}):
 cohort=[r for r in cases if (r["k"],r["g"])==kg]
 assert len(cohort)==15
 for candidate in ["fused2","tile8"]:
  reductions=[1-r["p50_ns"][candidate]/r["p50_ns"]["vec4"] for r in cohort]
  wins=sum(r["p50_ns"][candidate]*100<=r["p50_ns"]["vec4"]*95 for r in cohort)
  block_medians=[statistics.median([1-r["p50_ns"][candidate]/r["p50_ns"]["vec4"] for r in cohort if r["block"]==b]) for b in range(1,4)]
  accepted=wins>=14 and all(v>=.05 for v in block_medians)
  panels.append({"k":kg[0],"g":kg[1],"candidate":candidate,"reductions":reductions,"median_reduction":statistics.median(reductions),"min_reduction":min(reductions),"max_reduction":max(reductions),"block_median_reductions":block_medians,"runs_at_least_5_percent":wins,"repeatable_by_frozen_rule":accepted,"p50_ns_median":{c:statistics.median(r["p50_ns"][c] for r in cohort) for c in ["vec4",candidate]},"p95_ns_median":{c:statistics.median(r["p95_ns"][c] for r in cohort) for c in ["vec4",candidate]}})
summary={"schema":"flat.pvp3d-thor-repeatability/v1","protocol":"R2","preregistration_commit":(root/"preregistration-commit.txt").read_text().strip(),"source_revision":"50c0230cc645bd44f1520e2c7194878a4b13009c","device":"NVIDIA Tegra NVIDIA Thor","backend":"Vulkan","driver":"580.00","blocks":3,"invocations":len(runs),"raw_samples":sum(r["raw_samples"] for r in runs),"measured_rows":sum(r["measured_rows"] for r in runs),"skipped_rows":sum(r["skipped_rows"] for r in runs),"ANF_checks":288,"decision":"descriptive_fixed_rule_not_statistical_significance","promotion":"none","runs":runs,"panels":panels,"occupancy_counts":dict(collections.Counter((x["phase"]+":"+x["decision"].split(",")[0]) for x in ownership))}
(root/"summary.json").write_text(json.dumps(summary,indent=2)+"\n")
print(json.dumps({"invocations":len(runs),"raw_samples":summary["raw_samples"],"accepted":sum(p["repeatable_by_frozen_rule"] for p in panels),"decisions":len(panels),"occupancy":summary["occupancy_counts"]}))
